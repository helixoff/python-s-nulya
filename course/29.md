# Создаём своё API

## Минимальный сервис задач
Установи Flask в виртуальное окружение. Сохрани следующий код в отдельный файл api_app.py и запусти в терминале через python api_app.py. В книге длительные серверы автоматически останавливаются, поэтому этот пример запускается отдельно.
```python
from flask import Flask, jsonify, request

app = Flask(__name__)
tasks = {}
next_id = 1

@app.get("/tasks")
def list_tasks():
    return jsonify(list(tasks.values()))

@app.get("/tasks/<int:task_id>")
def get_task(task_id):
    task = tasks.get(task_id)
    if task is None:
        return jsonify(error="Задача не найдена"), 404
    return jsonify(task)

@app.post("/tasks")
def create_task():
    global next_id
    data = request.get_json(silent=True)
    if not isinstance(data, dict):
        return jsonify(error="Нужен JSON-объект"), 400
    title = data.get("title")
    if not isinstance(title, str) or not title.strip():
        return jsonify(error="Нужно непустое текстовое title"), 400
    task = {"id": next_id, "title": title.strip(), "done": False}
    tasks[next_id] = task
    next_id += 1
    return jsonify(task), 201

@app.patch("/tasks/<int:task_id>")
def update_task(task_id):
    if task_id not in tasks:
        return jsonify(error="Задача не найдена"), 404
    data = request.get_json(silent=True)
    if not isinstance(data, dict) or not isinstance(data.get("done"), bool):
        return jsonify(error="Нужно логическое поле done"), 400
    tasks[task_id]["done"] = data["done"]
    return jsonify(tasks[task_id])

@app.delete("/tasks/<int:task_id>")
def delete_task(task_id):
    if task_id not in tasks:
        return jsonify(error="Задача не найдена"), 404
    del tasks[task_id]
    return "", 204

if __name__ == "__main__":
    app.run(host="127.0.0.1", port=5000)
```
Пример хранит данные в памяти одного процесса. При перезапуске они исчезнут. Он предназначен для локального изучения, не для общей базы в многопроцессном сервере. Позже tasks можно заменить коллекцией MongoDB или таблицей SQLite.

## Как проверить
Браузер умеет открыть GET http://127.0.0.1:5000/tasks. Для POST и PATCH удобен отдельный клиент на Python:
```python
import requests
base = "http://127.0.0.1:5000"
created = requests.post(base + "/tasks", json={"title": "Пройти главу"}, timeout=3)
created.raise_for_status()
task = created.json()
print(task)
updated = requests.patch(base + f"/tasks/{task['id']}", json={"done": True}, timeout=3)
print(updated.status_code, updated.json())
```
Клиент запускается после старта сервера. Проверяй не только успешный путь: отправь пустое название, число вместо текста, неизвестный id, неверный тип done. Хороший API объясняет ошибку и возвращает соответствующий статус.

## Следующие шаги
Добавь постоянное хранение, разделение обработчиков и данных, проверку входных полей и тесты. Если приложение когда-нибудь выйдет за пределы компьютера, потребуются аутентификация и подходящий сервер размещения. Не включай debug для публичного сервера.

## Практика
1. Запусти API и создай две задачи.
2. Отметь одну выполненной и удали вторую.
3. Проверь ответы для отсутствующего id и пустого title.
4. Добавь фильтр /tasks?done=true. Явно разбери текстовый параметр, потому что bool("false") даёт True.
